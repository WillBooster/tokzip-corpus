```c++
  #include <locale>
  #include <fmt/core.h>

  int main() {
    std::locale::global(std::locale("ru_RU.UTF-8"));
    fmt::print("value = {}", 4.2);
  }
  ```

  prints \"value = 4.2\" regardless of the locale.

  For locale-specific formatting use the `n` specifier:

  ```c++
  std::locale::global(std::locale("ru_RU.UTF-8"));
  fmt::print("value = {:n}", 4.2);
  ```

  prints \"value = 4,2\".
